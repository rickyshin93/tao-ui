Creates resizable panes separated by draggable handles.

## When to use

Use Splitter for work surfaces where users need to allocate space between adjacent panes.

**Recommended**

- Set sensible min sizes.
- Persist sizes only when the layout is user-specific.

**Avoid**

- Do not use the component outside its intended interaction model.

### Basic usage

```tsx preview
import { Splitter } from '@rockshin/tao-ui';

function SplitterExample() {
  return (
```

### Three panels

Multiple panels share the available space; drag the adjacent separators to change their proportions.

```tsx preview
import { Splitter } from '@rockshin/tao-ui';

const Panel = Splitter.Panel;

const boxStyle = (bg: string): React.CSSProperties => ({
  display: 'flex',
  alignItems: 'center',
  justifyContent: 'center',
  height: '100%',
  minHeight: 40,
  background: bg,
  borderRadius: 4,
  color: 'var(--tao-color-text-secondary)',
  fontSize: 14,
});

export default function SplitterThreePanels() {
  return (
    <div
      style={{
        height: 200,
        border: '1px solid var(--tao-color-border-secondary)',
        borderRadius: 6,
      }}
    >
      <Splitter>
        <Panel defaultSize="25%" min={60}>
          <div style={boxStyle('var(--tao-color-fill-quaternary)')}>
            Sidebar
          </div>
        </Panel>
        <Panel min={100}>
          <div style={boxStyle('var(--tao-color-fill-tertiary)')}>Main</div>
        </Panel>
        <Panel defaultSize="25%" min={60}>
          <div style={boxStyle('var(--tao-color-fill-quaternary)')}>Detail</div>
        </Panel>
      </Splitter>
    </div>
  );
}
```

```tsx preview
      style={{
        height: 200,
```

### Controlled panel sizes

Observe size changes and keep the layout in application state so it can be reset.

```tsx preview
import { Splitter, Stack } from '@rockshin/tao-ui';

import { useState } from 'react';

const Panel = Splitter.Panel;

const boxStyle = (bg: string): React.CSSProperties => ({
  display: 'flex',
  alignItems: 'center',
  justifyContent: 'center',
  height: '100%',
  minHeight: 40,
  background: bg,
  borderRadius: 4,
  color: 'var(--tao-color-text-secondary)',
  fontSize: 14,
});

export default function SplitterControlled() {
  function Demo() {
    const [sizes, setSizes] = useState<number[]>([]);
    return (
      <Stack gap="sm">
        <p style={{ fontSize: 13, color: 'var(--tao-color-text-secondary)' }}>
          sizes: [{sizes.map((s) => Math.round(s)).join(', ')}]
        </p>
        <div
          style={{
            height: 200,
            border: '1px solid var(--tao-color-border-secondary)',
            borderRadius: 6,
          }}
        >
          <Splitter onResize={setSizes}>
            <Panel defaultSize="40%">
              <div style={boxStyle('var(--tao-color-fill-quaternary)')}>
                Left
              </div>
            </Panel>
            <Panel>
              <div style={boxStyle('var(--tao-color-fill-tertiary)')}>
                Right
              </div>
            </Panel>
          </Splitter>
        </div>
      </Stack>
    );
  }
  return <Demo />;
}
```

### Nested splitters

Combine horizontal and vertical splitters into an editor layout with bounded container heights.

```tsx preview
import { Splitter } from '@rockshin/tao-ui';

const Panel = Splitter.Panel;

const boxStyle = (bg: string): React.CSSProperties => ({
  display: 'flex',
  alignItems: 'center',
  justifyContent: 'center',
  height: '100%',
  minHeight: 40,
  background: bg,
  borderRadius: 4,
  color: 'var(--tao-color-text-secondary)',
  fontSize: 14,
});

export default function SplitterNested() {
  return (
    <div
      style={{
        height: 400,
        border: '1px solid var(--tao-color-border-secondary)',
        borderRadius: 6,
      }}
    >
      <Splitter>
        <Panel defaultSize="25%" collapsible min={0}>
          <div style={boxStyle('var(--tao-color-fill-quaternary)')}>
            Sidebar
          </div>
        </Panel>
        <Panel>
          <Splitter orientation="vertical">
            <Panel defaultSize="70%">
              <div style={boxStyle('var(--tao-color-fill-tertiary)')}>
                Editor
              </div>
            </Panel>
            <Panel collapsible min={0}>
              <div style={boxStyle('var(--tao-color-fill-quaternary)')}>
                Terminal
              </div>
            </Panel>
          </Splitter>
        </Panel>
      </Splitter>
    </div>
  );
}
```
