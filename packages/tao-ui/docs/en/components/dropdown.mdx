Opens a menu of actions or navigation choices from a trigger.

## When to use

Use Dropdown for compact action sets that do not need to stay visible.

**Recommended**

- Order frequent actions first.
- Use destructive styling only for destructive items.

**Avoid**

- Do not use Dropdown as a replacement for visible primary navigation.

### Trigger, arrow, and split button

```tsx preview
import { useState } from 'react';
import { Button, Dropdown, Stack } from '@rockshin/tao-ui';

const items = [
  { key: 'edit', label: 'Edit' },
  { key: 'duplicate', label: 'Duplicate' },
];

function DropdownTriggers() {
  const [feedback, setFeedback] = useState('');
  return (
    <>
      <Stack direction="horizontal" gap="md">
        <Dropdown trigger={['hover']} menu={{ items }}>
          <Button variant="secondary">Hover</Button>
        </Dropdown>
        <Dropdown trigger={['click']} arrow menu={{ items }}>
          <Button variant="secondary">Click with arrow</Button>
        </Dropdown>
        <Dropdown.Button menu={{ items }} onClick={() => setFeedback('save')}>
          Save
        </Dropdown.Button>
      </Stack>
      <output
        aria-live="polite"
        style={{
          display: 'block',
          marginTop: 12,
          whiteSpace: 'pre-wrap',
          overflowWrap: 'anywhere',
        }}
      >
        {feedback}
      </output>
    </>
  );
}

export default DropdownTriggers;
```

### Controlled menu selection

Manage selected keys in application state for view modes or filtering choices.

```tsx preview
import { Button, Dropdown, Stack } from '@rockshin/tao-ui';

import { useState } from 'react';

export default function DropdownSelectable() {
  const [single, setSingle] = useState<string[]>(['md']);
  const [multi, setMulti] = useState<string[]>(['bold']);
  return (
    <Stack direction="horizontal" gap="lg">
      <Dropdown
        menu={{
          selectable: true,
          selectedKeys: single,
          onSelect: ({ selectedKeys }) => setSingle(selectedKeys),
          items: [
            { key: 'sm', label: 'Small' },
            { key: 'md', label: 'Medium' },
            { key: 'lg', label: 'Large' },
          ],
        }}
      >
        <Button variant="secondary">Size: {single[0]}</Button>
      </Dropdown>

      <Dropdown
        menu={{
          selectable: true,
          multiple: true,
          selectedKeys: multi,
          onSelect: ({ selectedKeys }) => setMulti(selectedKeys),
          items: [
            { key: 'bold', label: 'Bold' },
            { key: 'italic', label: 'Italic' },
            { key: 'underline', label: 'Underline' },
          ],
        }}
      >
        <Button variant="secondary">Format ({multi.length})</Button>
      </Dropdown>
    </Stack>
  );
}
```

### Menu placement

Configure `placement`; the popup still avoids viewport edges.

```tsx preview
import { Button, Dropdown, type MenuItem } from '@rockshin/tao-ui';

const basicItems: MenuItem[] = [
  { key: 'profile', label: 'Profile' },
  { key: 'billing', label: 'Billing' },
  { key: 'settings', label: 'Settings' },
];

export default function DropdownPlacement() {
  return (
    <div
      style={{
        display: 'grid',
        gridTemplateColumns: 'repeat(3, auto)',
        gap: 12,
        width: 'fit-content',
      }}
    >
      {(
        [
          'topLeft',
          'top',
          'topRight',
          'bottomLeft',
          'bottom',
          'bottomRight',
        ] as const
      ).map((p) => (
        <Dropdown key={p} placement={p} menu={{ items: basicItems }}>
          <Button variant="secondary">{p}</Button>
        </Dropdown>
      ))}
    </div>
  );
}
```

### Trigger behavior

Try click and hover triggers; retain a click entry point for touch devices.

```tsx preview
import { Button, Dropdown, type MenuItem, Stack } from '@rockshin/tao-ui';

const basicItems: MenuItem[] = [
  { key: 'profile', label: 'Profile' },
  { key: 'billing', label: 'Billing' },
  { key: 'settings', label: 'Settings' },
];

export default function DropdownTrigger() {
  return (
    <Stack direction="horizontal" gap="lg">
      <Dropdown trigger={['hover']} menu={{ items: basicItems }}>
        <Button variant="secondary">Hover me</Button>
      </Dropdown>
      <Dropdown trigger={['click']} arrow menu={{ items: basicItems }}>
        <Button variant="secondary">Click (with arrow)</Button>
      </Dropdown>
    </Stack>
  );
}
```
