Provides a styled scroll container for overflow content.

## When to use

Use ScrollArea when a fixed region needs overflow behavior without changing page scroll.

**Recommended**

- Constrain height explicitly.
- Avoid hiding important actions below a small scroll region.

**Avoid**

- Do not use the component outside its intended interaction model.

### Basic usage

```tsx preview
import { ScrollArea } from '@rockshin/tao-ui';

function ScrollAreaExample() {
  return (
    <ScrollArea
      maxHeight={180}
      style={{
        border: '1px solid var(--tao-color-border-secondary)',
        borderRadius: 8,
      }}
    >
      <div style={{ padding: 16 }}>
        {Array.from({ length: 12 }, (_, index) => (
          <p key={index}>Activity item {index + 1}</p>
        ))}
      </div>
    </ScrollArea>
  );
}

export default ScrollAreaExample;
```

### Persistent scrollbars and custom size

```tsx preview
import { ScrollArea, Stack } from '@rockshin/tao-ui';

function ScrollbarStyles() {
  const content = Array.from({ length: 10 }, (_, index) => (
    <p key={index}>Scrollable row {index + 1}</p>
  ));

  return (
    <Stack direction="horizontal" gap="lg">
      <ScrollArea type="always" scrollbarSize={4} maxHeight={160}>
        <div style={{ padding: 12 }}>{content}</div>
      </ScrollArea>
      <ScrollArea type="always" scrollbarSize={10} maxHeight={160}>
        <div style={{ padding: 12 }}>{content}</div>
      </ScrollArea>
    </Stack>
  );
}

export default ScrollbarStyles;
```

### Horizontal scrolling

Constrain the available width and use the appropriate scrollbar orientation for horizontal content.

```tsx preview
import { ScrollArea, Tag } from '@rockshin/tao-ui';

const items = Array.from({ length: 50 }, (_, i) => `Item ${i + 1}`);

export default function ScrollAreaHorizontal() {
  return (
    <ScrollArea
      direction="horizontal"
      maxWidth={400}
      style={{
        border: '1px solid var(--tao-color-border-secondary)',
        borderRadius: 6,
      }}
    >
      <div
        style={{ display: 'flex', gap: 8, padding: 16, width: 'max-content' }}
      >
        {items.slice(0, 20).map((item) => (
          <Tag key={item} color="primary">
            {item}
          </Tag>
        ))}
      </div>
    </ScrollArea>
  );
}
```

### Both axes

Scroll each axis when content exceeds both dimensions without widening the page.

```tsx preview
import { ScrollArea } from '@rockshin/tao-ui';

const items = Array.from({ length: 50 }, (_, i) => `Item ${i + 1}`);

export default function ScrollAreaBoth() {
  return (
    <ScrollArea
      direction="both"
      maxHeight={200}
      maxWidth={300}
      style={{
        border: '1px solid var(--tao-color-border-secondary)',
        borderRadius: 6,
      }}
    >
      <div style={{ width: 600, padding: 16 }}>
        {items.map((item) => (
          <p
            key={item}
            style={{
              margin: '0 0 8px',
              color: 'var(--tao-color-text-secondary)',
              whiteSpace: 'nowrap',
            }}
          >
            {item} — This is a long line that overflows the container
            horizontally to demonstrate both scrollbars.
          </p>
        ))}
      </div>
    </ScrollArea>
  );
}
```

### Persistent scrollbars

`type="always"` retains the scrollbar track when scrollability should be more visible.

```tsx preview

const items = Array.from({ length: 50 }, (_, i) => `Item ${i + 1}`);

export default function ScrollAreaAlwaysVisible() {
  return (
    <ScrollArea
      type="always"
      maxHeight={200}
      style={{
        border: '1px solid var(--tao-color-border-secondary)',
        borderRadius: 6,
      }}
    >
      <div style={{ padding: 16 }}>
        {items.map((item) => (
          <p
            key={item}
            style={{
              margin: '0 0 8px',
              color: 'var(--tao-color-text-secondary)',
            }}
          >
            {item}
          </p>
        ))}
      </div>
    </ScrollArea>
  );
}
```

### Scrollbar size

Adjust scrollbar thickness while keeping the content area bounded.

```tsx preview
import { ScrollArea, Stack } from '@rockshin/tao-ui';

const items = Array.from({ length: 50 }, (_, i) => `Item ${i + 1}`);

export default function ScrollAreaCustomSize() {
  return (
    <Stack gap="lg">
      <div>
        <p
          style={{
            fontSize: 13,
            color: 'var(--tao-color-text-secondary)',
            marginBottom: 8,
          }}
        >
          Thin (4px)
        </p>
        <ScrollArea
          type="always"
          scrollbarSize={4}
          maxHeight={150}
          style={{
            border: '1px solid var(--tao-color-border-secondary)',
            borderRadius: 6,
          }}
        >
          <div style={{ padding: 16 }}>
            {items.slice(0, 30).map((item) => (
              <p
                key={item}
                style={{
                  margin: '0 0 8px',
                  color: 'var(--tao-color-text-secondary)',
                }}
              >
                {item}
              </p>
            ))}
          </div>
        </ScrollArea>
      </div>
      <div>
        <p
          style={{
            fontSize: 13,
            color: 'var(--tao-color-text-secondary)',
            marginBottom: 8,
          }}
        >
          Thick (10px)
        </p>
        <ScrollArea
          type="always"
          scrollbarSize={10}
          maxHeight={150}
          style={{
            border: '1px solid var(--tao-color-border-secondary)',
            borderRadius: 6,
          }}
        >
          <div style={{ padding: 16 }}>

              >
                {item}
              </p>

      </div>

}
```

```tsx preview
            key={item}
            style={{
              flexShrink: 0,

              height: 60,

              justifyContent: 'center',
```
