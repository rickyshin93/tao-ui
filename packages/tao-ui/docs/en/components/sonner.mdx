Displays temporary toast notifications with optional actions and provider-aware theming.

## When to use

Mount one Sonner near app root, then call toast() imperatively from event handlers. Supports toast.loading(), toast.promise(), and toast.custom() for richer async and custom notification flows.

**Recommended**

- Prefer short, status-oriented messages with one primary action at most.
- Enable richColors when semantic status color should be visible.

**Avoid**

- Do not use toast notifications for blocking confirmations.

## Examples

Each example runs independently, and notifications stay inside its preview. Applications normally mount one `Sonner` host.

```tsx preview="iframe-follow"
import { Button, Sonner, toast } from '@rockshin/tao-ui';

function SonnerExample() {
  return (
    <>
      <Button
        onClick={() =>

            title: 'Saved',
            description: 'Your profile has been updated.',
          })
        }
      >
        Show toast
      </Button>
      <Sonner />
    </>
  );
}

export default SonnerExample;
```

### Loading and promise lifecycle

```tsx preview="iframe-follow"
import { Button, Sonner, Stack, toast } from '@rockshin/tao-ui';

const wait = (duration: number) =>
  new Promise((resolve) => window.setTimeout(resolve, duration));

function AsyncToasts() {
  return (
    <>
      <Stack direction="horizontal" gap="sm">
        <Button
          onClick={() => {
            const id = toast.loading({ title: 'Syncing workspace' });
            window.setTimeout(() => {
              toast.dismiss(id);
              toast.success({ title: 'Workspace synced' });
            }, 1200);
          }}
        >
          Loading
        </Button>
        <Button
          variant="secondary"
          onClick={() =>
            toast.promise(wait(1200), {
              loading: { title: 'Publishing' },
              success: { title: 'Published' },
              error: { title: 'Publish failed' },
            })
          }
        >
          Promise
        </Button>
      </Stack>
      <Sonner richColors />
    </>
  );
}

export default AsyncToasts;
```

```tsx preview="iframe-follow"
        </Button>
```

```tsx preview="iframe-follow"
                title: `Queued task #${index}`,
```

### Submission feedback

A local delay simulates submission feedback for pending, success, and error states.

```tsx preview="iframe-follow"
import { Button, Sonner, Stack, toast } from '@rockshin/tao-ui';

function wait(ms: number) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

export default function SonnerFormSubmit() {
  return (
    <>
      <Stack direction="horizontal" gap="sm">
        <Button
          onClick={async () => {
            await toast.promise(wait(1200), {
              loading: {
                title: 'Submitting form',
                description: 'Validating fields and saving changes.',
              },
              success: {
                title: 'Profile updated',
                description: 'Name, email, and preferences are saved.',
              },
              error: {
                title: 'Submit failed',
              },
            });
          }}
        >
          Submit profile
        </Button>
        <Button
          variant="secondary"
          onClick={() =>
            toast
              .promise(
                wait(1200).then(() => {
                  throw new Error('Card declined');
                }),
                {
                  loading: {
                    title: 'Submitting payment',
                    description: 'Contacting payment gateway.',
                  },
                  success: { title: 'Paid' },
                  error: (error) => ({
                    title: 'Payment failed',
                    description: (error as Error).message,
                  }),
                },
              )
              .catch(() => undefined)
          }
        >
          Submit payment
        </Button>
      </Stack>
      <Sonner richColors />
    </>
  );
}
```
