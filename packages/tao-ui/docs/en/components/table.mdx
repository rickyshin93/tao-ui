Displays structured row and column data with selection, sorting, filtering, and pagination hooks.

## When to use

Use Table for dense, comparable records that benefit from aligned columns.

**Recommended**

- Keep columns focused on the task.
- Use stable row keys.
- Control sorting and pagination when data is remote.
- Gate secondary columns with column.responsive (e.g. ['lg']) instead of hand-rolled media queries.

**Avoid**

- Do not use Table for card-like content where comparison is not important.

### Fixed header, expandable rows, and summary

`scroll.y` caps the body height and pins the header; `expandable` renders details under a row (the expand column sits before the selection column); `summary` returns `Table.Summary.Row` / `Table.Summary.Cell` footer rows — `Cell.index` counts from the expand/selection columns to sync fixed columns, and `` pins the footer while the body scrolls.

```tsx preview
import { Table } from '@rockshin/tao-ui';

const data = Array.from({ length: 30 }, (_, index) => ({
  id: index + 1,
  name: 'User ' + (index + 1),
  team: ['Design', 'Engineering', 'Sales'][index % 3],
  score: 60 + ((index * 7) % 40),
}));

const columns = [
  {
    key: 'name',
    dataIndex: 'name' as const,
    title: 'Name',
    width: 160,
    fixed: 'left' as const,
  },
  { key: 'team', dataIndex: 'team' as const, title: 'Team', width: 240 },
  {
    key: 'score',
    dataIndex: 'score' as const,
    title: 'Score',
    width: 120,
    align: 'right' as const,
  },
];

function TableAdvanced() {
  return (
    <Table
      columns={columns}
      dataSource={data}
      rowKey="id"
      pagination={false}
      scroll={{ x: 640, y: 280 }}
      expandable={{
        expandedRowRender: (record) =>
          record.name + ' works in ' + record.team + '.',
        rowExpandable: (record) => record.team !== 'Sales',
      }}
      summary={(pageData) => (
        <Table.Summary fixed>
          <Table.Summary.Row>
            <Table.Summary.Cell index={0} colSpan={3}>
              Total
            </Table.Summary.Cell>
            <Table.Summary.Cell index={3} align="right">
              {pageData.reduce((sum, record) => sum + record.score, 0)}
            </Table.Summary.Cell>
          </Table.Summary.Row>
        </Table.Summary>
      )}
    />
  );
}

export default TableAdvanced;
```

### Sorting

Configure sorting on columns and click a header to switch the sort direction.

```tsx preview
import { Table, type TableColumn } from '@rockshin/tao-ui';

interface User {
  id: number;
  name: string;
  age: number;
  email: string;
  role: string;
  department: string;
}

const users: User[] = Array.from({ length: 46 }, (_, i) => ({
  id: i + 1,
  name: ['Alice', 'Bob', 'Charlie', 'Diana', 'Eve', 'Frank', 'Grace', 'Hank'][
    i % 8
  ],
  age: 22 + (i % 20),
  email: `user${i + 1}@example.com`,
  role: i % 3 === 0 ? 'admin' : i % 3 === 1 ? 'editor' : 'viewer',
  department: ['Engineering', 'Design', 'Marketing', 'Sales'][i % 4],
}));

export default function TableSorting() {
  const columns: TableColumn<User>[] = [
    {
      key: 'name',
      dataIndex: 'name',
      title: 'Name',
      sorter: (a, b) => a.name.localeCompare(b.name),
    },
    {
      key: 'age',
      dataIndex: 'age',
      title: 'Age',
      sorter: (a, b) => a.age - b.age,
      defaultSortOrder: 'ascend',
    },
    { key: 'role', dataIndex: 'role', title: 'Role' },
  ];
  return <Table columns={columns} dataSource={users} rowKey="id" />;
}
```

### Controlled row selection

Use a stable record ID for `rowKey`, such as a database ID. Keep selected keys in `rowSelection.selectedRowKeys` and update them from `rowSelection.onChange`.

Return `{ disabled: true }` from `rowSelection.getCheckboxProps(record)` for rows that cannot be selected. The header checkbox applies to eligible rows on the current page and preserves selection on other pages. Since [0.0.49](/guide/changelog), Table also uses the current disabled state when rows change and the callback stays the same.

```tsx preview
import { Stack, Table, type TableColumn } from '@rockshin/tao-ui';

import { useState } from 'react';

interface User {
  id: number;
  name: string;
  age: number;
  email: string;
  role: string;
  department: string;
}

const users: User[] = Array.from({ length: 46 }, (_, i) => ({
  id: i + 1,
  name: ['Alice', 'Bob', 'Charlie', 'Diana', 'Eve', 'Frank', 'Grace', 'Hank'][
    i % 8
  ],
  age: 22 + (i % 20),
  email: `user${i + 1}@example.com`,
  role: i % 3 === 0 ? 'admin' : i % 3 === 1 ? 'editor' : 'viewer',
  department: ['Engineering', 'Design', 'Marketing', 'Sales'][i % 4],
}));

const baseColumns: TableColumn<User>[] = [
  { key: 'id', dataIndex: 'id', title: 'ID', width: 60, align: 'center' },
  { key: 'name', dataIndex: 'name', title: 'Name' },
  { key: 'age', dataIndex: 'age', title: 'Age', width: 80 },
  { key: 'email', dataIndex: 'email', title: 'Email' },
  { key: 'role', dataIndex: 'role', title: 'Role' },
];

export default function TableRowSelection() {
  function Demo() {
    const [selectedKeys, setSelectedKeys] = useState<React.Key[]>([]);
    return (
      <Stack gap="sm">
        <p style={{ fontSize: 13, color: 'var(--tao-color-text-secondary)' }}>
          Selected:{' '}
          {selectedKeys.length === 0 ? 'none' : selectedKeys.join(', ')}
        </p>
        <Table
          columns={baseColumns}
          dataSource={users}
          rowKey="id"
          rowSelection={{
            selectedRowKeys: selectedKeys,
            onChange: (keys) => setSelectedKeys(keys),
            getCheckboxProps: (record) => ({
              disabled: record.role === 'viewer',
            }),
          }}
        />
      </Stack>
    );
  }
  return <Demo />;
}
```

</section>

### Responsive columns

Use responsive column settings on smaller screens while retaining identifying fields.

```tsx preview="iframe-follow"
import { Table } from '@rockshin/tao-ui';

interface User {
  id: number;
  name: string;
  age: number;
  email: string;
  role: string;
  department: string;
}

const users: User[] = Array.from({ length: 46 }, (_, i) => ({
  id: i + 1,
  name: ['Alice', 'Bob', 'Charlie', 'Diana', 'Eve', 'Frank', 'Grace', 'Hank'][
    i % 8
  ],
  age: 22 + (i % 20),
  email: `user${i + 1}@example.com`,
  role: i % 3 === 0 ? 'admin' : i % 3 === 1 ? 'editor' : 'viewer',
  department: ['Engineering', 'Design', 'Marketing', 'Sales'][i % 4],
}));

export default function TableResponsiveColumns() {
  return (
    <Table
      columns={[
        { key: 'name', dataIndex: 'name', title: 'Name' },
        {
          key: 'email',
          dataIndex: 'email',
          title: 'Email',
          responsive: ['md'],
        },
        {
          key: 'department',
          dataIndex: 'department',
          title: 'Department',
          responsive: ['lg'],
        },
      ]}
      dataSource={users.slice(0, 5)}
      rowKey="id"
      pagination={false}
    />
  );
}
```
