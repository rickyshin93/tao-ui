Single-line text input for short freeform values.

## When to use

Use Input for short text values such as names, filters, search terms, and identifiers.

**Recommended**

- Pair with FormField for labels and validation.
- Use type, inputMode, and autoComplete when applicable.

**Avoid**

- Do not use the component outside its intended interaction model.

### Controlled value and reset

Keep value and onChange in application state; reset updates the same state.

```tsx preview
import { useState } from 'react';
import { Input, Button, Stack } from '@rockshin/tao-ui';
export default function Example() {
  const [value, setValue] = useState('tao-ui');
  return (
    <Stack direction="vertical">
      <Input
        aria-label="Project name"
        value={value}
        onChange={(e) => setValue(e.target.value)}
      />
      <Button onClick={() => setValue('')}>Clear</Button>
      <output>{value || 'Empty'}</output>
    </Stack>
  );
}
```

### Password visibility

Toggle the native type with a named button that does not submit the form.

```tsx preview
import { useState } from 'react';
import { Input, Button, Stack } from '@rockshin/tao-ui';
export default function Example() {
  const [visible, setVisible] = useState(false);
  return (
    <Stack direction="vertical">
      <Input
        aria-label="Password"
        type={visible ? 'text' : 'password'}
        defaultValue="example-password"
        autoComplete="new-password"
      />
      <Button aria-pressed={visible} onClick={() => setVisible(!visible)}>
        {visible ? 'Hide password' : 'Show password'}
      </Button>
    </Stack>
  );
}
```

### Search on submit

Use a native form for Enter and button submission, with results shown inline.

```tsx preview
import { useState } from 'react';
import { Input, Button, Stack } from '@rockshin/tao-ui';
export default function Example() {
  const [query, setQuery] = useState('');
  return (
    <form
      onSubmit={(e) => {
        e.preventDefault();
        setQuery(String(new FormData(e.currentTarget).get('query')));
      }}
    >
      <Stack direction="vertical">
        <Input
          name="query"
          type="search"
          aria-label="Search components"
          placeholder="Search components"
          required
        />
        <Button type="submit">Search</Button>
        <output aria-live="polite">
          {query ? `Searching for: ${query}` : 'Enter a component name'}
        </output>
      </Stack>
    </form>
  );
}
```

</section>

### Focus with a ref

The ref points to the native input for restoring focus after another action.

```tsx preview
import { useRef } from 'react';
import { Input, Button, Stack } from '@rockshin/tao-ui';
export default function Example() {
  const ref = useRef<HTMLInputElement>(null);
  return (
    <Stack direction="vertical">
      <Input
        ref={ref}
        aria-label="Focus target"
        defaultValue="Select this text"
      />
      <Button
        onClick={() => {
          ref.current?.focus();
          ref.current?.select();
        }}
      >
        Focus and select
      </Button>
    </Stack>
  );
}
```
