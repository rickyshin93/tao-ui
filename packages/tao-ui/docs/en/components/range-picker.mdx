Lets users select a start and end date as one range value.

## When to use

Use RangePicker for reporting windows, booking windows, and filters with start/end dates.

**Recommended**

- Use presets for common ranges such as today, this week, or last 30 days.
- Make inclusive/exclusive range behavior clear when it matters.

**Avoid**

- Do not use the component outside its intended interaction model.

### Controlled range

```tsx preview
import { RangePicker } from '@rockshin/tao-ui';
import dayjs, { type Dayjs } from 'dayjs';
import { useState } from 'react';

function ControlledRangePicker() {
  const [value, setValue] = useState<[Dayjs | null, Dayjs | null] | null>([
    dayjs().startOf('week'),
    dayjs().endOf('week'),
  ]);

  return <RangePicker value={value} onChange={setValue} />;
}

export default ControlledRangePicker;
```

### Range selection and clearing

Choose the start and end dates in sequence, maintain the range value, and allow clearing.

```tsx preview
import { FormField, RangePicker, Stack } from '@rockshin/tao-ui';

import type { Dayjs } from 'dayjs';

import { useState } from 'react';

export default function DatePickerRangePickerBasic() {
  function Demo() {
    const [val, setVal] = useState<[Dayjs | null, Dayjs | null] | null>(null);
    return (
      <Stack gap="md" style={{ maxWidth: 500 }}>
        <FormField label="Date Range">
          <RangePicker value={val} onChange={(d) => setVal(d)} />
        </FormField>
        <p style={{ fontSize: 13, color: 'var(--tao-color-text-secondary)' }}>
          Range: {val?.[0]?.format('YYYY-MM-DD') ?? '?'} ~{' '}
          {val?.[1]?.format('YYYY-MM-DD') ?? '?'}
        </p>
      </Stack>
    );
  }
  return <Demo />;
}
```

### Range presets

Offer common report ranges using the same value structure as manual selections.

```tsx preview
import { FormField, RangePicker, Stack } from '@rockshin/tao-ui';

import type { Dayjs } from 'dayjs';

import dayjs from 'dayjs';

import { useState } from 'react';

export default function DatePickerRangePickerPresets() {
  function Demo() {
    const [val, setVal] = useState<[Dayjs | null, Dayjs | null] | null>(null);
    return (
      <Stack gap="md" style={{ maxWidth: 500 }}>
        <FormField label="Range with presets">
          <RangePicker
            value={val}
            onChange={(d) => setVal(d)}
            presets={[
              {
                label: 'This week',
                value: [dayjs().startOf('week'), dayjs().endOf('week')],
              },
              {
                label: 'This month',
                value: [dayjs().startOf('month'), dayjs().endOf('month')],
              },
              {
                label: 'Last 7 days',
                value: [dayjs().subtract(7, 'day'), dayjs()],
              },
              {
                label: 'Last 30 days',
                value: [dayjs().subtract(30, 'day'), dayjs()],
              },
            ]}
          />
        </FormField>
      </Stack>
    );
  }
  return <Demo />;
}
```

### Locale switching

TaoProvider locale updates built-in actions, placeholders, and calendar labels.

```tsx preview
import { useState } from 'react';
import {
  RangePicker,
  TaoProvider,
  zhCN,
  enUS,
  Button,
  Stack,
} from '@rockshin/tao-ui';
export default function Example() {
  const [chinese, setChinese] = useState(false);
  return (
    <Stack direction="vertical">
      <Button onClick={() => setChinese(!chinese)}>
        {chinese ? 'Switch to English' : '切换到中文'}
      </Button>
      <TaoProvider locale={chinese ? zhCN : enUS}>
        <RangePicker />
      </TaoProvider>
    </Stack>
  );
}
```

### Controlled value and clearing

Pass `value` to control the range and save the value received by `onChange`. Set `value` to `null` to reset the range from your application.

With both endpoints enabled, the built-in clear control reports `onChange(null, ['', ''])`. Since [0.0.49](/guide/changelog), RangePicker also discards any text still being edited without first emitting it as a selected range.

```tsx preview
import { useState } from 'react';
import type { Dayjs } from 'dayjs';
import { RangePicker, Button, Stack } from '@rockshin/tao-ui';
export default function Example() {
  const [value, setValue] = useState<[Dayjs | null, Dayjs | null] | null>(null);
  return (
    <Stack direction="vertical">
      <RangePicker value={value} onChange={setValue} allowClear />
      <Button onClick={() => setValue(null)}>Clear value</Button>
      <output>
        {value?.map((date) => date?.format('YYYY-MM-DD') || '—').join(' → ') ||
          'No selection'}
      </output>
    </Stack>
  );
}
```

</section>
