Provides global component defaults, portal scoping, and theme seed token overrides.

## When to use

Wrap an app or subtree with TaoProvider to set size, disabled, variant, and theme defaults.

**Recommended**

- Prefer provider-level defaults for app-wide decisions.
- Use nested providers for scoped theme overrides.

**Avoid**

- Do not override derived CSS variables directly unless a theme token cannot express the change.

### Brand colors

Change brand colors across the same components while retaining status semantics.

```tsx preview
import {
  Button,
  Checkbox,
  Input,
  Select,
  Stack,
  Switch,
  Tag,
  TaoProvider,
} from '@rockshin/tao-ui';

import { type CSSProperties } from 'react';

const swatchLabel: CSSProperties = {
  fontSize: 12,
  color: 'var(--tao-color-text-tertiary)',
  minWidth: 72,
  fontFamily: 'var(--tao-font-family-code)',
};

function ControlRow({
  label,
  selectOptions = [
    { value: '1', label: 'Option 1' },
    { value: '2', label: 'Option 2' },
  ],
}: {
  label: string;
  selectOptions?: { value: string; label: string }[];
}) {
  return (
    <Stack direction="horizontal" gap="sm" align="center">
      <span style={swatchLabel}>{label}</span>
      <Button variant="primary">Primary</Button>
      <Button variant="secondary">Secondary</Button>
      <Input placeholder="Input" style={{ width: 140 }} />
      <Select options={selectOptions} style={{ width: 140 }} />
      <Switch defaultChecked />
      <Checkbox defaultChecked>Check</Checkbox>
      <Tag color="primary" variant="solid">
        Tag
      </Tag>
    </Stack>
  );
}

const BRAND_PALETTES = [
  { name: 'Default (Anthropic-neutral)', color: 'oklch(0.205 0.01 70)' },
  { name: 'Vercel Black', color: 'oklch(0.145 0 0)' },
  { name: 'Linear Indigo', color: 'oklch(0.52 0.19 275)' },
  { name: 'Stripe Purple', color: 'oklch(0.55 0.22 300)' },
  { name: 'Supabase Green', color: 'oklch(0.62 0.17 155)' },
  { name: 'Notion Orange', color: 'oklch(0.68 0.17 45)' },
];

export default function ProviderBrandColors() {
  return (
    <Stack gap="lg">
      {BRAND_PALETTES.map((p) => (
        <TaoProvider key={p.name} theme={{ token: { colorPrimary: p.color } }}>
          <Stack gap="sm">
            <span style={swatchLabel}>{p.name}</span>
            <ControlRow label={p.color.slice(0, 22)} />
          </Stack>
        </TaoProvider>
      ))}
    </Stack>
  );
}
```

```tsx preview
      size="large"

                inner · orange, size inherited (large)
```

### Right-to-left layouts

Set `direction="rtl"` on TaoProvider or use HTML `dir="rtl"` on a surrounding element. Components inherit the local direction, and popups keep the direction of their trigger even when rendered into a portal. An explicit component `direction` prop or nested provider can override it; Carousel uses `rtl`. Locale and direction are independent. Explicit popup placements such as `bottomLeft` retain their physical meaning.

```tsx preview
import { Select, Stack, TaoProvider } from '@rockshin/tao-ui';

export default function DirectionExample() {
  return (
    <TaoProvider direction="rtl">
      <Stack gap="md">
        <Select defaultValue="ar" options={[{ value: 'ar', label: 'العربية' }, { value: 'en', label: 'English' }]} />
        <Select direction="ltr" defaultValue="en" options={[{ value: 'en', label: 'English override' }]} />
      </Stack>
    </TaoProvider>
  );
}
```
