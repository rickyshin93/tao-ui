A styled numeric input built on tao-ui number formatting and validation behavior.

## When to use

Use InputNumber for numeric values that need min, max, step, precision, or formatted display.

**Recommended**

- Set min and max when the domain is bounded.
- Use the headless API when custom rendering is required.

**Avoid**

- Do not use a numeric input for identifiers such as account numbers.

### Basic usage

```tsx preview
import { InputNumber } from '@rockshin/tao-ui';
import { useState } from 'react';

function InputNumberExample() {
  const [value, setValue] = useState('1500');

  return (
    <InputNumber
      value={value}
      onChange={setValue}
      prefix="$"
      suffix="USD"
      decimalPlaces={2}
      thousandsSeparator=","
    />
  );
}

export default InputNumberExample;
```

### Currency formatting

Thousands separators, decimals, and a prefix format display while retaining a usable value.

```tsx preview
import { useState } from 'react';

import { InputNumber } from '@rockshin/tao-ui';

function ControlledWrapper(props: React.ComponentProps<typeof InputNumber>) {
  const [value, setValue] = useState((props.value as string | undefined) ?? '');
  return (
    <div
      style={{
        display: 'flex',
        flexDirection: 'column',
        gap: 12,
        minWidth: 0,
        width: '100%',
        maxWidth: 320,
      }}
    >
      <InputNumber
        {...props}
        value={value}
        onChange={(v) => {
          setValue(v);
          props.onChange?.(v);
        }}
      />
      <code
        style={{
          background: 'var(--tao-color-bg-container)',
          padding: 8,
          borderRadius: 4,
        }}
      >
        actual value: {JSON.stringify(value)}
      </code>
    </div>
  );
}

export default function InputNumberCurrency() {
  const args: React.ComponentProps<typeof InputNumber> = {
    prefix: '€',
    thousandsSeparator: ',',
    decimalSeparator: '.',
    decimalPlaces: 2,
    value: '150000',
    placeholder: 'Amount',
  };
  return <ControlledWrapper {...args} />;
}
```

## Regional formatting

Regional grouping, decimal symbols and digits inherit `TaoProvider.locale.numberLocale`; set `locale="fr-FR"` on this component to override it. Explicit separators take precedence. Numeric input keeps canonical ASCII strings and precision while displaying localized digits.
