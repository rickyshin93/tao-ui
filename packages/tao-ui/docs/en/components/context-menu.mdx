Displays contextual actions from a right click or long press interaction.

## When to use

Use ContextMenu for secondary actions attached to an object or region.

**Recommended**

- Keep items short and action-oriented.
- Provide another reachable path for critical actions.

**Avoid**

- Do not put primary workflows only inside a context menu.

### Groups and submenus

Organize actions with groups and nested `children`; mark destructive actions with `danger`. Right-click the area.

```tsx preview
import { ContextMenu } from '@rockshin/tao-ui';

function Area({ children = 'Right-click here' }: { children?: string }) {
  return (
    <div
      style={{
        display: 'grid',
        placeItems: 'center',
        width: 320,
        height: 160,
        border: '1px dashed var(--tao-color-border)',
        borderRadius: 8,
        color: 'var(--tao-color-text-tertiary)',
        fontSize: 13,
        userSelect: 'none',
      }}
    >
      {children}
    </div>
  );
}

export default function ContextMenuGroupsAndSubmenu() {
  return (
    <ContextMenu
      menu={{
        items: [
          {
            type: 'group',
            label: 'View',
            children: [
              { key: 'zoom-in', label: 'Zoom in' },
              { key: 'zoom-out', label: 'Zoom out' },
            ],
          },
          { type: 'divider' },
          {
            key: 'share',
            label: 'Share',
            children: [
              { key: 'mail', label: 'Email' },
              { key: 'link', label: 'Copy link' },
            ],
          },
          { key: 'rename', label: 'Rename' },
          { key: 'delete', label: 'Delete', danger: true },
        ],
      }}
    >
      <Area />
    </ContextMenu>
  );
}
```

### Selectable menu items

Keep selection in `selectedKeys` and `onSelect`, for example for view preferences.

```tsx preview
import { ContextMenu } from '@rockshin/tao-ui';

import { useState } from 'react';

function Area({ children = 'Right-click here' }: { children?: string }) {
  return (
    <div
      style={{
        display: 'grid',
        placeItems: 'center',
        width: 320,
        height: 160,
        border: '1px dashed var(--tao-color-border)',
        borderRadius: 8,
        color: 'var(--tao-color-text-tertiary)',
        fontSize: 13,
        userSelect: 'none',
      }}
    >
      {children}
    </div>
  );
}

export default function ContextMenuSelectable() {
  const [checked, setChecked] = useState<string[]>(['grid']);
  return (
    <ContextMenu
      menu={{
        selectable: true,
        multiple: true,
        selectedKeys: checked,
        onSelect: ({ selectedKeys }) => setChecked(selectedKeys),
        items: [
          { key: 'grid', label: 'Show grid' },
          { key: 'ruler', label: 'Show ruler' },
          { key: 'guides', label: 'Show guides' },
        ],
      }}
    >
      <Area>{`Right-click — ${checked.length} shown`}</Area>
    </ContextMenu>
  );
}
```

### Disabled context menu

`disabled` turns off the custom menu. Keep essential actions available outside a context menu.

```tsx preview
import { ContextMenu, type MenuItem } from '@rockshin/tao-ui';

function Area({ children = 'Right-click here' }: { children?: string }) {
  return (
    <div
      style={{
        display: 'grid',
        placeItems: 'center',
        width: 320,
        height: 160,
        border: '1px dashed var(--tao-color-border)',
        borderRadius: 8,
        color: 'var(--tao-color-text-tertiary)',
        fontSize: 13,
        userSelect: 'none',
      }}
    >
      {children}
    </div>
  );
}

export default function ContextMenuDisabled() {
  return (
    <ContextMenu
      menu={{
        items: [
          { key: 'undo', label: 'Undo', extra: '⌘Z' },
          { key: 'redo', label: 'Redo', disabled: true, extra: '⇧⌘Z' },
          { type: 'divider' },
          { key: 'select-all', label: 'Select all', extra: '⌘A' },
        ] as MenuItem[],
      }}
    >
      <Area />
    </ContextMenu>
  );
}
```
