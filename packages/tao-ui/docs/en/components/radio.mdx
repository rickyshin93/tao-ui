Selects exactly one option from a mutually exclusive set.

## When to use

Use RadioGroup when every option is visible and only one can be selected.

**Recommended**

- Use concise labels.
- Provide a default value when one option is recommended or required.

**Avoid**

- Do not use RadioGroup for very long option lists; use Select instead.

### Basic usage

```tsx preview
import { RadioGroup } from '@rockshin/tao-ui';
import { useState } from 'react';

function RadioExample() {
  const [value, setValue] = useState('pro');

  return (
    <RadioGroup
      value={value}
      onChange={setValue}
      options={[
        { label: 'Free', value: 'free' },
        { label: 'Pro', value: 'pro' },
        { label: 'Enterprise', value: 'enterprise', disabled: true },
      ]}
    />
  );
}

export default RadioExample;
```

### Standalone states

```tsx preview
import { Radio, RadioGroup, Stack } from '@rockshin/tao-ui';
import { useState } from 'react';

function RadioStates() {
  const [value, setValue] = useState('selected');

  return (
    <RadioGroup value={value} onChange={setValue}>
      <Stack direction="horizontal" gap="lg">
        <Radio value="selected">Selected</Radio>
        <Radio value="idle">Unselected</Radio>
        <Radio value="disabled" disabled>
          Disabled
        </Radio>
      </Stack>
    </RadioGroup>
  );
}

export default RadioStates;
```

### Controlled radio group

Group onChange returns the selected value directly.

```tsx preview
import { useState } from 'react';
import { RadioGroup, Stack } from '@rockshin/tao-ui';
export default function Example() {
  const [value, setValue] = useState('Weekly');
  return (
    <Stack direction="vertical">
      <RadioGroup
        options={['Daily', 'Weekly', 'Monthly']}
        value={value}
        onChange={setValue}
      />
      <output>Frequency: {value}</output>
    </Stack>
  );
}
```

### Native form submission

Set a group name to read its selected value from FormData.

```tsx preview
import { useState } from 'react';
import { RadioGroup, Button, Stack } from '@rockshin/tao-ui';
export default function Example() {
  const [result, setResult] = useState('');
  return (
    <form
      onSubmit={(e) => {
        e.preventDefault();
        setResult(String(new FormData(e.currentTarget).get('plan')));
      }}
    >
      <Stack direction="vertical">
        <RadioGroup name="plan" options={['Free', 'Pro']} defaultValue="Free" />
        <Button type="submit">Read plan</Button>
        <output>{result}</output>
      </Stack>
    </form>
  );
}
```

### Radio group sizes

Group options inherit size for consistent alignment and targets.

```tsx preview
import { RadioGroup, Stack } from '@rockshin/tao-ui';
export default function Example() {
  return (
    <Stack direction="vertical">
      {(['small', 'medium', 'large'] as const).map((size) => (
        <RadioGroup
          key={size}
          size={size}
          options={['A', 'B', 'C']}
          defaultValue="A"
        />
      ))}
    </Stack>
  );
}
```
