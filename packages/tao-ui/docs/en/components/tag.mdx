Displays compact labels, statuses, categories, or checkable chips.

## When to use

Use Tag for lightweight metadata or status markers. Use CheckableTag for compact filter choices.

**Recommended**

- Use semantic colors consistently.
- Keep tag text short.

**Avoid**

- Do not encode critical state by color alone.

### Custom colors and closable tags

```tsx preview
import { useState } from 'react';
import { Stack, Tag } from '@rockshin/tao-ui';

function CustomTags() {
  const [feedback, setFeedback] = useState('');
  return (
    <>
      <Stack direction="horizontal" gap="sm">
        <Tag color="#245b57" variant="solid">
          Custom teal
        </Tag>
        <Tag color="#8a5a21" variant="outlined">
          Custom ochre
        </Tag>
        {feedback !== 'closed' && (
          <Tag color="primary" closeIcon onClose={() => setFeedback('closed')}>
            Closable
          </Tag>
        )}
        <Tag disabled color="primary" variant="solid">
          Disabled
        </Tag>
      </Stack>
      <output
        aria-live="polite"
        style={{
          display: 'block',
          marginTop: 12,
          whiteSpace: 'pre-wrap',
          overflowWrap: 'anywhere',
        }}
      >
        {feedback}
      </output>
    </>
  );
}

export default CustomTags;
```

### Checkable tags

```tsx preview
import { CheckableTag, Stack } from '@rockshin/tao-ui';
import { useState } from 'react';

function CheckableTags() {
  const [selected, setSelected] = useState(['Design']);

  return (
    <Stack direction="horizontal" gap="sm">
      {['Design', 'Engineering', 'Research'].map((label) => (
        <CheckableTag
          key={label}
          checked={selected.includes(label)}
          onChange={(checked) =>
            setSelected((items) =>
              checked
                ? [...items, label]
                : items.filter((item) => item !== label),
            )
          }
        >
          {label}
        </CheckableTag>
      ))}
    </Stack>
  );
}

export default CheckableTags;
```

### Checkable tags

Use `checked` and its change callback for lightweight category filtering.

```tsx preview
import { CheckableTag, Stack } from '@rockshin/tao-ui';

import { useState } from 'react';

export default function TagCheckable() {
  function Demo() {
    const [selected, setSelected] = useState<string[]>(['react']);
    const options = ['react', 'vue', 'angular', 'svelte', 'solid'];
    return (
      <Stack direction="horizontal" gap="sm">
        {options.map((opt) => (
          <CheckableTag
            key={opt}
            checked={selected.includes(opt)}
            onChange={(checked) =>
              setSelected((prev) =>
                checked ? [...prev, opt] : prev.filter((x) => x !== opt),
              )
            }
          >
            {opt}
          </CheckableTag>
        ))}
      </Stack>
    );
  }
  return <Demo />;
}
```
